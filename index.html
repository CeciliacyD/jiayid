                if (currentUploadTarget.startsWith('gallery')) {

                    const index = currentUploadTarget.split('-')[1];

                    const item = document.querySelector(`.gallery-item[data-index="${index}"]`);

                    item.style.backgroundImage = imgUrl;

                    item.querySelector('span').style.display = 'none';

                } else if (currentUploadTarget === 'avatar-academic') {

                    const avatar = document.getElementById('avatar-academic');

                    avatar.style.backgroundImage = imgUrl;

                    avatar.querySelector('span').style.display = 'none';

                } else if (currentUploadTarget === 'avatar-personal') {

                    const avatar = document.getElementById('avatar-personal');

                    avatar.style.backgroundImage = imgUrl;

                    avatar.querySelector('span').style.display = 'none';

                }


                fileInput.value = '';

                currentUploadTarget = null;

            };

            reader.readAsDataURL(file);

        });
